    function updateActivePromotionBanner() {
      const banner = document.getElementById('activeVoucherBanner');
      const txt = document.getElementById('appliedVoucherCodeText');
      if (!banner || !txt) return;
      const parts = [];
      if (activeVoucher) parts.push(`Mã ${activeVoucher.code} (-${activeVoucher.discountAmt.toLocaleString()}đ)`);
      if (spinBonusDiscount > 0) parts.push(`Quay Trợ Giá (-${spinBonusDiscount.toLocaleString()}đ)`);
      if (tradeInCredit > 0) parts.push(`Thu Cũ Đổi Mới (-${tradeInCredit.toLocaleString()}đ)`);
      if (parts.length > 0) {
        banner.style.display = 'block';
        txt.innerText = parts.join(' + ');
      } else {
        banner.style.display = 'none';
      }
    }

    window.calcFuelSavings = function() {
      const km = parseInt(document.getElementById('kmRangeSlider')?.value || '25');
      document.getElementById('kmPerDayLabel').innerText = `${km} km / ngày`;
      const gasYear = Math.round(km * 365 * 0.022 * 24000) + 450000;
      const evYear = Math.round(km * 365 * 0.012 * 3000);

      const saved3Y = saved1Y * 3;
      document.getElementById('fuelSavingsResultBox').innerHTML = `
        • Chi phí xe xăng: <b style="color:#dc2626;">~${gasYear.toLocaleString()}đ/năm</b> | Xe điện Hồng Vinh: <b style="color:#16a34a;">~${evYear.toLocaleString()}đ/năm</b><br>

    function getEffectiveVehicle(v) {
      const ov = vehicleOverrides[v.id] || {};
      return {
        ...v,
        livePrice: ov.livePrice ? Number(ov.livePrice) : v.livePrice,
        oldPrice: ov.oldPrice ? Number(ov.oldPrice) : v.oldPrice,
        stockLeft: ov.stockLeft !== undefined ? Number(ov.stockLeft) : (v.stockLeft ?? 3)
      };
    }

    function computeCustomerPrice(v) {
      const eff = getEffectiveVehicle(v);
      const voucherOff = activeVoucher ? Math.min(activeVoucher.discountAmt, Math.round(eff.livePrice * 0.12)) : 0;
      const afterVoucher = Math.max(0, eff.livePrice - voucherOff - spinBonusDiscount);
      return afterVoucher;
    }

    function getModalNetPrice() {
      if (!currentModalVehicle) return 0;
      let base = computeCustomerPrice(currentModalVehicle);
      const nano = document.getElementById('upsellNano')?.checked ? 150000 : 0;
      const helmet = document.getElementById('upsellHelmetLock')?.checked ? 190000 : 0;
      return Math.max(0, base + nano + helmet - tradeInCredit);
    }

    window.recalcInstallmentPlan = function() {
      if (!currentModalVehicle) return;
      const finalPrice = getModalNetPrice();
      document.getElementById('modalFinalPrice').innerText = finalPrice.toLocaleString() + 'đ';
      const downPct = parseInt(document.getElementById('calcDownPaymentPct')?.value || '20');
      const months = parseInt(document.getElementById('calcMonthsTerm')?.value || '9');
      const downAmt = Math.round(finalPrice * downPct / 100);
      const loanAmt = finalPrice - downAmt;
      const monthlyPay = Math.round((loanAmt / months) + (loanAmt * 0.013));
      document.getElementById('calcInstallmentResultText').innerHTML = `
        👉 Trả trước: <b>${downAmt.toLocaleString()}đ</b> | Góp mỗi tháng: <b style="color:#dc2626; font-size:0.95rem;">~${monthlyPay.toLocaleString()}đ / tháng</b> (trong ${months} tháng)
      `;
    };

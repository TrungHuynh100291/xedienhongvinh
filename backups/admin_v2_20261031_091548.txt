    // TINH TOAN THUE HO KINH DOANH CA THE (MAU 01/CNKD - TT 40/2021/TT-BTC)
    function renderTaxAccountingDashboard() {
      const orders = Object.values(allOrdersMap).filter(o => !(o.status || '').includes('Hủy'));
      let goodsRev = 0;
      let serviceRev = 0;
      let passThroughTotal = 0;

      orders.forEach(o => {
        const amt = Number(o.finalPrice || 0);
        const srv = Number(o.serviceFee || 0);
        const pass = Number(o.passThroughFee || 0);
        passThroughTotal += pass;
        if (o.taxCategory === 'SERVICE_7') {
          serviceRev += (amt + srv);
        } else {
          goodsRev += amt;
          serviceRev += srv;
        }
      });

      const goodsVat = Math.round(goodsRev * 0.01);
      const goodsPit = Math.round(goodsRev * 0.005);
      const goodsTaxTotal = goodsVat + goodsPit;

      const serviceVat = Math.round(serviceRev * 0.05);
      const servicePit = Math.round(serviceRev * 0.02);
      const serviceTaxTotal = serviceVat + servicePit;

      const totalRev = goodsRev + serviceRev;
      const totalTax = goodsTaxTotal + serviceTaxTotal;

      document.getElementById('kpiTotalRev').innerText = totalRev.toLocaleString() + 'đ';
      document.getElementById('kpiGoodsRev').innerText = goodsRev.toLocaleString() + 'đ';
      document.getElementById('kpiGoodsTaxAmt').innerText = `Thuế GTGT (1%) + TNCN (0.5%): ${goodsTaxTotal.toLocaleString()}đ`;
      document.getElementById('kpiServiceRev').innerText = serviceRev.toLocaleString() + 'đ';
      document.getElementById('kpiServiceTaxAmt').innerText = `Thuế GTGT (5%) + TNCN (2%): ${serviceTaxTotal.toLocaleString()}đ`;
      document.getElementById('kpiTotalTaxPayable').innerText = totalTax.toLocaleString() + 'đ';
      document.getElementById('kpiPassThroughExcluded').innerText = `Thu hộ tách riêng (0% thuế): ${passThroughTotal.toLocaleString()}đ`;

      const tbody = document.getElementById('taxSummaryTableBody');
      if (tbody) {
        tbody.innerHTML = `
          <tr>
            <td><b>1. Phân phối, cung cấp hàng hóa</b> (Bán xe máy điện, xe đạp điện, bình ắc quy)</td>
            <td>[28]</td>
            <td><b>${goodsRev.toLocaleString()}đ</b></td>
            <td>${goodsVat.toLocaleString()}đ (1.0%)</td>
            <td>${goodsPit.toLocaleString()}đ (0.5%)</td>
            <td><b style="color:#16a34a;">${goodsTaxTotal.toLocaleString()}đ</b></td>
          </tr>
          <tr>
            <td><b>2. Dịch vụ, xây dựng không bao thầu NVL</b> (Công sửa chữa, bảo dưỡng, cứu hộ)</td>
            <td>[29]</td>
            <td><b>${serviceRev.toLocaleString()}đ</b></td>
            <td>${serviceVat.toLocaleString()}đ (5.0%)</td>
            <td>${servicePit.toLocaleString()}đ (2.0%)</td>
            <td><b style="color:#b45309;">${serviceTaxTotal.toLocaleString()}đ</b></td>
          </tr>
          <tr style="background:#fef2f2; font-weight:900;">
            <td>TỔNG CỘNG PHẢI NỘP TỜ KHAI 01/CNKD</td>
            <td>[32]</td>
            <td>${totalRev.toLocaleString()}đ</td>
            <td>${(goodsVat + serviceVat).toLocaleString()}đ</td>
            <td>${(goodsPit + servicePit).toLocaleString()}đ</td>
            <td style="color:#dc2626; font-size:1rem;">${totalTax.toLocaleString()}đ</td>
          </tr>
        `;
      }
    }

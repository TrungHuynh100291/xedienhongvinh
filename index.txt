      const now = new Date();
      const mins = now.getHours() * 60 + now.getMinutes();
      const inNoonSlot = (mins >= (11 * 60 + 30)) && (mins <= (13 * 60 + 30));
      const inNightSlot = (mins >= (19 * 60 + 30)) && (mins <= (21 * 60 + 30));

    function getEffectiveVehicle(v) {
      const ov = vehicleOverrides[v.id] || {};
      return {
        ...v,
        livePrice: ov.livePrice ? Number(ov.livePrice) : v.livePrice,
        oldPrice: ov.oldPrice ? Number(ov.oldPrice) : v.oldPrice,
        stockStatus: ov.stockStatus || v.stockStatus || 'IN_STOCK',
        stockLeft: ov.stockLeft !== undefined ? Number(ov.stockLeft) : (v.stockLeft ?? 3),
        restockDays: ov.restockDays !== undefined ? Number(ov.restockDays) : (v.restockDays ?? 3)
      };
    }

    function computeCustomerPrice(v) {
      const eff = getEffectiveVehicle(v);
      const voucherOff = activeVoucher ? Math.min(activeVoucher.discountAmt, Math.round(eff.livePrice * 0.12)) : 0;
      const preorderBonus = eff.stockStatus === 'PREORDER' ? 300000 : 0;
      return Math.max(0, eff.livePrice - voucherOff - spinBonusDiscount - preorderBonus);
    }

    window.renderVehiclesGrid = function() {

        if (currentPersona === 'preorder') {
          if (v.stockStatus !== 'PREORDER' && v.stockLeft > 0) return false;
        } else if (currentPersona !== 'all' && v.persona && v.persona !== currentPersona) {
          return false;
        }
        if (kw && !v.name.toLowerCase().includes(kw) && !v.brand.toLowerCase().includes(kw)) return false;
        return true;

    function getModalNetPrice() {
      if (!currentModalVehicle) return 0;
      let base = computeCustomerPrice(currentModalVehicle);
      const nano = document.getElementById('upsellNano')?.checked ? 150000 : 0;
      const helmet = document.getElementById('upsellHelmetLock')?.checked ? 190000 : 0;
      return Math.max(0, base + nano + helmet - tradeInCredit);
    }

    window.recalcInstallmentPlan = function() {
      if (!currentModalVehicle) return;
      const finalPrice = getModalNetPrice();
      document.getElementById('modalFinalPrice').innerText = finalPrice.toLocaleString() + 'đ';
      const downPct = parseInt(document.getElementById('calcDownPaymentPct')?.value || '20');
      const months = parseInt(document.getElementById('calcMonthsTerm')?.value || '9');
      const downAmt = Math.round(finalPrice * downPct / 100);
      const loanAmt = finalPrice - downAmt;
      const monthlyPay = Math.round((loanAmt / months) + (loanAmt * 0.013));
      document.getElementById('calcInstallmentResultText').innerHTML = `
        👉 Trả trước: <b>${downAmt.toLocaleString()}đ</b> | Góp mỗi tháng: <b style="color:#dc2626;">~${monthlyPay.toLocaleString()}đ / tháng</b> (${months} tháng)
      `;
    };
